cmake_minimum_required(VERSION 3.20)
project(FFmpegLite VERSION 0.2.0 LANGUAGES C CXX)

option(FFLITE_BUILD_EXAMPLES "Build examples" ON)
option(FFLITE_BUILD_DOC_EXAMPLES "Build complete examples from docs/" OFF)
include(CTest)
include(cmake/FFmpeg.cmake)

add_library(fflite src/types.cpp src/frame.cpp src/codec.cpp src/processor.cpp src/container.cpp)
add_library(fflite::fflite ALIAS fflite)
target_compile_features(fflite PUBLIC cxx_std_17)
set_target_properties(fflite PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(fflite PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(fflite PRIVATE fflite_ffmpeg)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(fflite PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(FFLITE_BUILD_EXAMPLES)
  add_executable(fflite_decode examples/decode.cpp)
  target_link_libraries(fflite_decode PRIVATE fflite::fflite)
  add_executable(fflite_memory examples/memory.cpp)
  target_link_libraries(fflite_memory PRIVATE fflite::fflite)
endif()

if(BUILD_TESTING)
  add_executable(fflite_core_tests tests/core.cpp)
  target_link_libraries(fflite_core_tests PRIVATE fflite::fflite)
  foreach(case memory_video ownership video_processor audio_processor audio_codec containers failures callback_forms)
    add_test(NAME ${case} COMMAND fflite_core_tests ${case})
  endforeach()
endif()

if(FFLITE_BUILD_DOC_EXAMPLES)
  add_subdirectory(docs/examples)
endif()
